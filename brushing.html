cars = transpose(cars_long)

{
// First create a filter for brushing
const filter = Plot.select("brush");

// Create the first plot (Horsepower vs MPG)
const plot1 = Plot.plot({
  marks: [
    Plot.dot(cars, {
      x: "displ",
      y: "city",
      stroke: "manufacturer",
      fill: filter
    })
  ],
  color: {
    legend: true,
    label: "manufacturer"
  }
})

// Create the second plot (Weight vs MPG)
const plot2 = Plot.plot({
  marks: [
    Plot.dot(cars, {
      x: "hwy",
      y: "cty",
      stroke: "manufacturer",
      fill: filter
    })
  ],
  color: {
    legend: true,
    label: "manufacturer"
  }
})

// Combine plots into a panel
Plot.plot({
  marks: [
    Plot.frame({x: 0, y: 0, width: 400, height: 300}, [plot1]),
    Plot.frame({x: 400, y: 0, width: 400, height: 300}, [plot2])
  ],
  style: {
    width: 800,
    height: 300
  }
})
}

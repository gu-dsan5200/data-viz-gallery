The grammar of graphics

We’re familiar with the grammar of graphics from our familiarity with ggplot2. These principles have also been applied to develop the semantics of several Javascript plotting libraries, including Vega-Lite and Observable Plot. Let’s see how the grammar is implemented.

We’ll start with some data from the US Energy Information Administration, looking at energy consumption over time of different energy sources.

Code 
energy = transpose(energy_long).map(
    item => {
        item.year = d3.utcFormat("%Y")(new Date(item.year, 0,1));// convert to date 
        return item;
    }
)
Inputs.table(energy,{
    header: {
        year: "Year",
        type: "Source",
        quadrillion_btu : "Energy use (Quadrillion BTU)"
    }
})

Let’s start with some familiar visualizations:

Code 
Plot.plot({
    title: "Energy use by source, 1973-2022",
    x: { label: "Year", type:'time'},
    y: { label: "Energy (Quadrillion BTU)"},
    color: {scheme: "Tableau10", legend: true, label: "Source", columns: 1,
        style: {float: "right"}
    },
    stroke: {scheme: "Tableau10", legend: true, label: "Source", columns: 5},
    style: {
        backgroundColor: "white",
        color: "#145da0",
        fontFamily: "Atkinson Hyperlegible",
        fontSize: "14px",
        padding: "16px"
    },
    marks: [
        Plot.dot(energy, 
            {x: 'year',
            y: "quadrillion_btu", 
            fill: "type",
            tip: true,
            }),
        Plot.line(energy, 
            {x: 'year',
            y: "quadrillion_btu", 
            stroke: "type"}),
        Plot.frame(), // add box around plot
    ]
})

{
    const points = vl.markPoint()

const lines = vl.markLine()
    .data(energy)
    .encode(

        vl.y().fieldQ('quadrillion_btu').axis({title: "Energy use (Quadrillion BTU)"}),
        vl.color().fieldN('type')
    ).width(400).height(400);
return vl.layer(lines, points).render();
}
